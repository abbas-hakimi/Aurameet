import AppError from "../utils/AppError.js";
import { registerUser, loginUser, refreshAccessToken, logoutUser } from "../services/auth.service.js";

const validateCredentials = ({ name, username, password }, registering = false) => {
    if (registering && (typeof name !== "string" || !name.trim() || name.trim().length > 80)) throw new AppError("Name must be 1 to 80 characters", 400);
    if (typeof username !== "string" || !/^[a-z0-9_.]{3,30}$/i.test(username.trim())) throw new AppError("Username must be 3 to 30 letters, numbers, dots or underscores", 400);
    if (typeof password !== "string" || password.length < 8 || Buffer.byteLength(password, "utf8") > 72) throw new AppError("Password must be 8 to 72 bytes", 400);
};

export const register = async (req, res) => {
    validateCredentials(req.body, true);
    const data = await registerUser(req.body);
    res.status(201).json({ success: true, message: "Registration successful", data });
};
export const login = async (req, res) => {
    validateCredentials(req.body);
    const data = await loginUser(req.body);
    res.json({ success: true, message: "Login successful", data });
};
export const getMe = (req, res) => res.json({ success: true, data: { user: req.user } });
export const refresh = async (req, res) => {
    if (typeof req.body.refreshToken !== "string" || !req.body.refreshToken) throw new AppError("Refresh token required", 401);
    const accessToken = await refreshAccessToken(req.body.refreshToken);
    res.json({ success: true, data: { accessToken } });
};
export const logout = async (req, res) => {
    if (typeof req.body.refreshToken !== "string") throw new AppError("Refresh token required", 400);
    await logoutUser(req.body.refreshToken);
    res.json({ success: true, message: "Logged out successfully" });
};

import jwt from "jsonwebtoken";
import env from "../config/env.js";
import User from "../models/User.js";

const authenticate = async (req, res, next) => {
    try {
        const authHeader = req.headers.authorization;

        if (!authHeader || !authHeader.startsWith("Bearer ")) {
            return res.status(401).json({
                success: false,
                message: "Authentication required"
            });
        }

        const token = authHeader.split(" ")[1];

        const decoded = jwt.verify(
            token,
            env.jwtAccessSecret
        );

        const user = await User.findById(decoded.sub);

        if (!user) {
            return res.status(401).json({
                success: false,
                message: "User no longer exists"
            });
        }

        req.user = user;

        next();

    } catch (error) {

        return res.status(401).json({
            success: false,
            message: "Invalid or expired access token"
        });
    }
};

export default authenticate;

export const errorHandler = (
    err,
    req,
    res,
    next
) => {
    let statusCode = err.statusCode || 500;
    let message = statusCode < 500 ? err.message : "Internal server error";
    if (err?.code === 11000) {
        statusCode = 409;
        message = "A record with that value already exists";
    } else if (err?.name === "CastError" || err?.name === "ValidationError") {
        statusCode = 400;
        message = "Invalid request data";
    } else if (err?.type === "entity.parse.failed") {
        statusCode = 400;
        message = "Invalid JSON request body";
    } else if (err?.type === "entity.too.large") {
        statusCode = 413;
        message = "Request body is too large";
    }
    if (statusCode >= 500) console.error(err);

    res.status(statusCode).json({
        success: false,
        message
    });
};

import jwt from "jsonwebtoken";
import env from "../config/env.js";
import User from "../models/User.js";

export const socketAuth = async (socket, next) => {

    try {

        const token =
            socket.handshake.auth?.token;

        if (!token) {
            return next(
                new Error("Authentication required")
            );
        }

        const decoded = jwt.verify(
            token,
            env.jwtAccessSecret
        );

        const user = await User.findById(
            decoded.sub
        );

        if (!user) {
            return next(
                new Error("User not found")
            );
        }

        socket.user = {
            id: user._id.toString(),
            name: user.name,
            username: user.username,
            avatar: user.avatar,
            role: user.role
        };

        next();

    } catch (error) {

        next(
            new Error("Invalid or expired access token")
        );
    }
};

import bcrypt from "bcrypt";

import User from "../models/User.js";
import AppError from "../utils/AppError.js";
import {
    generateAccessToken,
    generateRefreshToken,
    verifyRefreshToken
} from "../utils/token.js";

export const registerUser = async ({ name, username, password }) => {

    name = name.trim();
    username = username.trim().toLowerCase();

    const existingUser = await User.findOne({ username });

    if (existingUser) {
        throw new AppError("Username already exists", 409);
    }

    const hashedPassword = await bcrypt.hash(password, 10);

    const user = await User.create({
        name,
        username,
        password: hashedPassword
    });

    const accessToken = generateAccessToken(user);
    const refreshToken = generateRefreshToken(user);

    user.refreshToken = refreshToken;
    await user.save();

    return {
        user: {
            id: user._id,
            name: user.name,
            username: user.username,
            role: user.role
        },
        accessToken,
        refreshToken
    };
};


export const loginUser = async ({ username, password }) => {

    username = username.trim().toLowerCase();
    const user = await User.findOne({ username }).select("+password +refreshToken");

    if (!user) {
        throw new AppError("Invalid username or password", 401);
    }

    const passwordMatched = await bcrypt.compare(
        password,
        user.password
    );

    if (!passwordMatched) {
        throw new AppError("Invalid username or password", 401);
    }

    user.lastLogin = new Date();
    await user.save();

    const accessToken = generateAccessToken(user);
    const refreshToken = generateRefreshToken(user);

    user.refreshToken = refreshToken;
    await user.save();

    return {
        user: {
            id: user._id,
            name: user.name,
            username: user.username,
            role: user.role
        },
        accessToken,
        refreshToken
    };
};

export const refreshAccessToken = async (refreshToken) => {
    let decoded;
    try { decoded = verifyRefreshToken(refreshToken); }
    catch { throw new AppError("Invalid or expired refresh token", 401); }

    const user = await User.findById(decoded.sub).select("+refreshToken");

    if (!user) {
        throw new AppError("Invalid or expired refresh token", 401);
    }

    if (user.refreshToken !== refreshToken) throw new AppError("Invalid refresh token", 401);

    const accessToken = generateAccessToken(user);

    return accessToken;
};

export const logoutUser = async (refreshToken) => {
    try {
        const decoded = verifyRefreshToken(refreshToken);

        const user = await User.findById(decoded.sub).select("+refreshToken");

        if (user && user.refreshToken === refreshToken) {
            user.refreshToken = null;
            await user.save();
        }
    } catch {
        // Token is already invalid/expired
    }
};

import crypto from "crypto";
import mongoose from "mongoose";
import Meeting from "../models/Meeting.js";
import Participant from "../models/Participant.js";
import AppError from "../utils/AppError.js";

const settingKeys = ["waitingRoom", "lockMeeting", "allowChat", "allowScreenShare"];
const codeFilter = (code) => {
    if (typeof code !== "string" || !/^[a-z0-9]{8}$/i.test(code)) throw new AppError("Invalid meeting code", 400);
    return code.toUpperCase();
};
const findMeeting = async (code) => {
    const meeting = await Meeting.findOne({ meetingCode: codeFilter(code) });
    if (!meeting) throw new AppError("Meeting not found", 404);
    return meeting;
};
const requireHost = (meeting, userId, action) => {
    if (meeting.host.toString() !== userId.toString()) throw new AppError(`Only the host can ${action}`, 403);
};

export const createMeeting = async ({ title, hostId, scheduledAt, settings = {} }) => {
    if (typeof title !== "string" || !title.trim() || title.trim().length > 120) throw new AppError("Meeting title must be 1 to 120 characters", 400);
    if (scheduledAt !== undefined && (typeof scheduledAt !== "string" || !Number.isFinite(Date.parse(scheduledAt)))) throw new AppError("Invalid scheduled date", 400);
    validateSettings(settings);
    let meetingCode;
    do { meetingCode = crypto.randomBytes(4).toString("hex").toUpperCase(); }
    while (await Meeting.exists({ meetingCode }));
    return Meeting.create({ meetingCode, title: title.trim(), host: hostId, scheduledAt, settings });
};

const validateSettings = (settings) => {
    if (!settings || typeof settings !== "object" || Array.isArray(settings) || Object.keys(settings).some(k => !settingKeys.includes(k)) || Object.values(settings).some(v => typeof v !== "boolean")) throw new AppError("Invalid meeting settings", 400);
};

export const getMeetingByCode = async (meetingCode, userId) => {
    const meeting = await Meeting.findOne({ meetingCode: codeFilter(meetingCode) }).populate("host", "name username avatar");
    if (!meeting) throw new AppError("Meeting not found", 404);
    if (meeting.host._id.toString() !== userId.toString()) meeting.waitingParticipants = undefined;
    return meeting;
};

export const joinMeeting = async ({ meetingCode, userId, name }) => {
    const meeting = await findMeeting(meetingCode);
    if (meeting.status === "ended") throw new AppError("Meeting has ended", 409);
    if (meeting.status === "scheduled") throw new AppError("Meeting has not started yet", 409);
    const isHost = meeting.host.toString() === userId.toString();
    const active = await Participant.findOne({ meeting: meeting._id, user: userId, leftAt: null });
    if (active) return { meeting, waiting: false, participant: active };
    if (meeting.settings.lockMeeting && !isHost) throw new AppError("Meeting is locked", 409);
    if (meeting.settings.waitingRoom && !isHost) {
        await Meeting.updateOne({ _id: meeting._id, "waitingParticipants.user": { $ne: userId } }, { $push: { waitingParticipants: { user: userId, name, requestedAt: new Date() } } });
        const freshMeeting = await Meeting.findById(meeting._id);
        return { meeting: freshMeeting, waiting: true, participant: null };
    }
    if (meeting.waitingParticipants.some(w => w.user.toString() === userId.toString())) {
        await Meeting.updateOne({ _id: meeting._id }, { $pull: { waitingParticipants: { user: userId } } });
        meeting.waitingParticipants = meeting.waitingParticipants.filter(w => w.user.toString() !== userId.toString());
    }
    const participant = await Participant.findOneAndUpdate(
        { meeting: meeting._id, user: userId, leftAt: null },
        { $setOnInsert: { name, role: isHost ? "host" : "participant", joinedAt: new Date() } },
        { upsert: true, new: true, setDefaultsOnInsert: true }
    );
    return { meeting, waiting: false, participant };
};

export const getUserMeetings = (userId) => Meeting.find({ host: userId }).sort({ createdAt: -1 }).populate("host", "name username");

export const startMeeting = async (meetingCode, userId) => {
    const meeting = await findMeeting(meetingCode);
    requireHost(meeting, userId, "start the meeting");
    if (meeting.status !== "scheduled") throw new AppError(meeting.status === "live" ? "Meeting is already live" : "Meeting has already ended", 409);
    meeting.status = "live";
    meeting.startedAt = new Date();
    await meeting.save();
    return meeting;
};

export const endMeeting = async (meetingCode, userId) => {
    const meeting = await findMeeting(meetingCode);
    requireHost(meeting, userId, "end the meeting");
    if (meeting.status !== "live") throw new AppError("Only a live meeting can be ended", 409);
    meeting.status = "ended";
    meeting.endedAt = new Date();
    meeting.waitingParticipants = [];
    await meeting.save();
    await Participant.updateMany({ meeting: meeting._id, leftAt: null }, { $set: { leftAt: meeting.endedAt } });
    return meeting;
};

    const meeting = await findMeeting(meetingCode);

        { $setOnInsert: { name: waiting.name, role: "participant", joinedAt: new Date() } },

    );
};

import Meeting from "../models/Meeting.js";
import Message from "../models/Message.js";

export const registerChatSocket = (io, socket) => {
    let windowStarted = Date.now();
    let sends = 0;
    const safe = (event, fn) => async (payload) => {
        try { await fn(payload && typeof payload === "object" && !Array.isArray(payload) ? payload : {}); }
        catch { socket.emit(event, { message: "Unable to complete chat request" }); }
    };
    socket.on("chat:send", safe("chat:error", async ({ meetingCode, message }) => {
        const code = typeof meetingCode === "string" ? meetingCode.toUpperCase() : "";
        if (!code || socket.data.meetingCode !== code || typeof message !== "string" || !message.trim() || message.trim().length > 2000) throw new Error("Invalid chat message");
        const now = Date.now();
        if (now - windowStarted > 10000) { windowStarted = now; sends = 0; }
        if (++sends > 20) throw new Error("Too many messages");
        const meeting = await Meeting.findOne({ meetingCode: code });
        if (!meeting || meeting.status !== "live") throw new Error("Chat is only available during a live meeting");
        if (!meeting.settings.allowChat && meeting.host.toString() !== socket.user.id) throw new Error("Chat is disabled");
        const saved = await Message.create({ meeting: meeting._id, sender: socket.user.id, message: message.trim(), type: "user" });
        const chatMessage = { id: saved._id.toString(), sender: { id: socket.user.id, name: socket.user.name, username: socket.user.username, avatar: socket.user.avatar }, message: saved.message, timestamp: saved.createdAt };
        io.to(code).emit("chat:message", chatMessage);
    }));

    socket.on("chat:history", safe("chat:error", async ({ meetingCode }) => {
        const code = typeof meetingCode === "string" ? meetingCode.toUpperCase() : "";
        if (!code || socket.data.meetingCode !== code) return;
        const meeting = await Meeting.findOne({ meetingCode: code });
        if (!meeting) return;
        if (meeting.status !== "live" || (!meeting.settings.allowChat && meeting.host.toString() !== socket.user.id)) throw new Error("Chat is unavailable");
        const docs = await Message.find({ meeting: meeting._id }).sort({ createdAt: -1 }).limit(100).populate("sender", "name username avatar").lean();
        socket.emit("chat:history", docs.reverse().map(msg => ({
            id: msg._id.toString(),
            sender: msg.sender ? { id: msg.sender._id.toString(), name: msg.sender.name, username: msg.sender.username, avatar: msg.sender.avatar } : null,
            message: msg.message,
            timestamp: msg.createdAt
        })));
    }));
};

import { joinMeeting, admitParticipant, removeParticipant, removeWaitingParticipant, updateMeetingSettings, leaveMeeting as persistLeave } from "../services/meeting.service.js";
import Meeting from "../models/Meeting.js";

const meetingRooms = new Map();
const safeHandler = (socket, eventName, fn) => async (payload) => {
    try { await fn(payload && typeof payload === "object" && !Array.isArray(payload) ? payload : {}); }
    catch (error) { socket.emit(eventName, { message: error.statusCode && error.statusCode < 500 ? error.message : "Unable to complete request" }); }
};
const notifyWaiting = (io, hostId, event, payload) => io.to(`user:${hostId}`).emit(event, payload);
const clearWaiting = async (io, socket, exceptCode) => {
    const previous = socket.data.waitingMeetingCode;
    if (!previous || previous === exceptCode) return;
    const meeting = await Meeting.findOneAndUpdate({ meetingCode: previous }, { $pull: { waitingParticipants: { user: socket.user.id } } }, { new: true });
    if (meeting) notifyWaiting(io, meeting.host.toString(), "waiting:left", { userId: socket.user.id, meetingCode: previous });
    await socket.leave(`waiting:${previous}`);
    socket.data.waitingMeetingCode = null;
};
const addSocketToMeeting = async (io, socket, code) => {
    if (socket.data.meetingCode && socket.data.meetingCode !== code) await removeSocketFromMeeting(io, socket);
    await clearWaiting(io, socket, code);
    if (socket.data.waitingMeetingCode) {
        const oldWaitingCode = socket.data.waitingMeetingCode;
        await socket.leave(`waiting:${oldWaitingCode}`);
        socket.data.waitingMeetingCode = null;
    }
    socket.join(code);
    socket.data.meetingCode = code;
    socket.data.waitingMeetingCode = null;
    if (!meetingRooms.has(code)) meetingRooms.set(code, new Map());
    const room = meetingRooms.get(code);
    room.set(socket.id, { socketId: socket.id, user: socket.user ?? socket.data.user });
    return [...room.values()];
};

export const registerMeetingSocket = (io, socket) => {
    socket.on("meeting:join", safeHandler(socket, "meeting:error", async ({ meetingCode }) => {
        if (typeof meetingCode !== "string") throw new Error("Invalid meeting code");
        const result = await joinMeeting({ meetingCode, userId: socket.user.id, name: socket.user.name });
        const code = result.meeting.meetingCode;
        if (result.waiting) {
            await clearWaiting(io, socket, code);
            socket.data.waitingMeetingCode = code;
            await socket.join(`waiting:${code}`);
            socket.emit("waiting:status", { status: "waiting", meetingCode: code, message: "Waiting for the host to admit you" });
            const waiting = result.meeting.waitingParticipants.find(w => w.user.toString() === socket.user.id);
            const hostId = result.meeting.host.toString();
            notifyWaiting(io, hostId, "waiting:participant", { user: socket.user, requestedAt: waiting?.requestedAt });
            return;
        }
        const participants = await addSocketToMeeting(io, socket, code);
        socket.emit("meeting:participants", participants);
        socket.to(code).emit("participant:joined", { socketId: socket.id, user: socket.user });
        if (result.meeting.host.toString() === socket.user.id) socket.emit("waiting:list", result.meeting.waitingParticipants);
    }));

    socket.on("meeting:leave", safeHandler(socket, "meeting:error", async () => removeSocketFromMeeting(io, socket)));

    socket.on("host:admit", safeHandler(socket, "meeting:error", async ({ meetingCode, participantUserId }) => {
        const participant = await admitParticipant(meetingCode, socket.user.id, participantUserId);
        await admitWaitingSockets(io, meetingCode.toUpperCase(), participantUserId, participant);
    }));

    socket.on("host:reject", safeHandler(socket, "meeting:error", async ({ meetingCode, participantUserId }) => {
        await removeWaitingParticipant(meetingCode, socket.user.id, participantUserId);
        const code = meetingCode.toUpperCase();
        const targets = await io.in(`user:${participantUserId}`).fetchSockets();
        for (const target of targets) if (target.data.waitingMeetingCode === code) {
            await target.leave(`waiting:${code}`);
            target.data.waitingMeetingCode = null;
            target.emit("waiting:rejected", { message: "The host did not admit you" });
        }
    }));

    socket.on("host:remove", safeHandler(socket, "meeting:error", async ({ meetingCode, participantUserId }) => {
        await removeParticipant(meetingCode, socket.user.id, participantUserId);
        const code = meetingCode.toUpperCase();
        const targets = await io.in(`user:${participantUserId}`).fetchSockets();
        for (const target of targets) if (target.data.meetingCode === code) {
            target.emit("participant:removed", { message: "You were removed by the host" });
            await removeSocketFromMeeting(io, target, true);
        }
        io.to(code).emit("participant:removed", { userId: participantUserId });
    }));

    socket.on("meeting:lock", safeHandler(socket, "meeting:error", async ({ meetingCode, locked }) => {
        if (typeof locked !== "boolean") throw new Error("locked must be a boolean");
        const meeting = await updateMeetingSettings(meetingCode, socket.user.id, { lockMeeting: locked });
        io.to(meeting.meetingCode).emit("meeting:settings-updated", { settings: meeting.settings });
    }));

    socket.on("screenshare:start", safeHandler(socket, "meeting:error", async ({ meetingCode }) => {
        if (socket.data.meetingCode !== meetingCode?.toUpperCase()) throw new Error("Join the meeting first");
        const meeting = await Meeting.findOne({ meetingCode: socket.data.meetingCode });
        if (!meeting || meeting.status !== "live") throw new Error("Meeting is not live");
        if (!meeting.settings.allowScreenShare && meeting.host.toString() !== socket.user.id) throw new Error("Screen sharing is disabled");
        socket.to(socket.data.meetingCode).emit("screenshare:started", { socketId: socket.id });
    }));
    socket.on("screenshare:stop", safeHandler(socket, "meeting:error", async () => {
        if (socket.data.meetingCode) socket.to(socket.data.meetingCode).emit("screenshare:stopped", { socketId: socket.id });
    }));

    const relay = (event, key) => (payload) => {
        try {
            const data = payload && typeof payload === "object" ? payload : {};
            const code = socket.data.meetingCode;
            if (!code || typeof data.to !== "string" || !data[key] || !meetingRooms.get(code)?.has(data.to)) return;
            io.to(data.to).emit(event, { from: socket.id, [key]: data[key] });
        } catch { socket.emit("meeting:error", { message: "Invalid signaling payload" }); }
    };
    socket.on("webrtc:offer", relay("webrtc:offer", "offer"));
    socket.on("webrtc:answer", relay("webrtc:answer", "answer"));
    socket.on("webrtc:ice-candidate", relay("webrtc:ice-candidate", "candidate"));
};

export const admitWaitingSockets = async (io, code, userId, participant) => {
    const targets = await io.in(`user:${userId}`).fetchSockets();
    for (const target of targets) if (target.data.waitingMeetingCode === code) {
        const participants = await addSocketToMeeting(io, target, code);
        target.emit("waiting:admitted", { meetingCode: code, participant });
        target.emit("meeting:participants", participants);
        io.to(code).emit("participant:joined", { socketId: target.id, user: target.data.user });
    }
};

export const closeMeetingRoom = async (io, code) => {
    io.to(code).emit("meeting:ended", { meetingCode: code });
    io.to(`waiting:${code}`).emit("meeting:ended", { meetingCode: code });
    const sockets = await io.in(code).fetchSockets();
    for (const socket of sockets) await removeSocketFromMeeting(io, socket, true);
    const waitingSockets = await io.in(`waiting:${code}`).fetchSockets();
    for (const socket of waitingSockets) {
        socket.data.waitingMeetingCode = null;
        await socket.leave(`waiting:${code}`);
    }
    meetingRooms.delete(code);
};

export const removeSocketFromMeeting = async (io, socket, skipPersistence = false) => {
    const user = socket.user ?? socket.data.user;
    if (!user?.id) return;
    const code = socket.data.meetingCode;
    const waitingCode = socket.data.waitingMeetingCode;
    if (waitingCode) {
        const otherWaitingSockets = await io.in(`user:${user.id}`).fetchSockets();
        const hasOtherWaitingSocket = otherWaitingSockets.some(other => other.id !== socket.id && other.data.waitingMeetingCode === waitingCode);
        socket.data.waitingMeetingCode = null;
        if (!hasOtherWaitingSocket) {
            const meeting = await Meeting.findOneAndUpdate({ meetingCode: waitingCode }, { $pull: { waitingParticipants: { user: user.id } } }, { new: true });
            if (meeting) notifyWaiting(io, meeting.host.toString(), "waiting:left", { userId: user.id, meetingCode: waitingCode });
        }
        await socket.leave(`waiting:${waitingCode}`);
    }
    if (!code) return;
    const room = meetingRooms.get(code);
    room?.delete(socket.id);
    socket.leave(code);
    socket.data.meetingCode = null;
    io.to(code).except(socket.id).emit("participant:left", { socketId: socket.id, userId: user.id });
    const stillHere = room && [...room.values()].some(p => p.user.id === user.id);
    if (room?.size === 0) meetingRooms.delete(code);
    if (!skipPersistence && !stillHere) {
        try { await persistLeave(code, user.id); } catch (err) { if (err.statusCode !== 409) console.error("Unable to persist participant leave", err.message); }
    }
};

import { Server } from "socket.io";
import { socketAuth } from "../middleware/socketAuth.js";

import {
    registerMeetingSocket,
    removeSocketFromMeeting
} from "./meeting.socket.js";

import {
    registerChatSocket
} from "./chat.socket.js";


export const initializeSocket = (server, env) => {

    const io = new Server(server, {
        cors: {
            origin: env.clientUrl,
            methods: ["GET", "POST"],
            credentials: true
        }
    });

    io.use(socketAuth);

    io.on("connection", (socket) => {
        socket.data.meetingCode = null;
        socket.data.waitingMeetingCode = null;
        socket.data.user = socket.user;
        socket.join(`user:${socket.user.id}`);

        console.log(
            `Socket connected: ${socket.id}`
        );

        registerMeetingSocket(io, socket);

        registerChatSocket(io, socket);

        socket.on("disconnect", () => {

            removeSocketFromMeeting(io, socket).catch((error) => {
                console.error("Socket disconnect cleanup failed:", error.message);
            });

            console.log(
                `Socket disconnected: ${socket.id}`
            );
        });
    });

    return io;
};
